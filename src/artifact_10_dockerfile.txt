# Multi-stage Dockerfile for Daejin Market Engine Daemon
FROM node:20-slim AS builder

WORKDIR /usr/src/app

COPY package*.json ./
COPY tsconfig.json ./
RUN npm ci

COPY src/ ./src
COPY target/idl/ ./target/idl/
RUN npm run build

FROM node:20-slim AS runner
WORKDIR /app

ENV NODE_ENV=production

COPY package*.json ./
RUN npm ci --only=production

COPY --from=builder /usr/src/app/dist ./dist
COPY --from=builder /usr/src/app/target ./target

USER node

EXPOSE 8080
CMD ["node", "dist/orchestrator.js"]
